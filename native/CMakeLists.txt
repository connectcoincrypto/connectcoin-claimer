cmake_minimum_required(VERSION 3.22)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(connectcoin_claimer_native VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(FetchContent)
find_package(Threads REQUIRED)
set(ENABLE_PROGRAMS OFF CACHE BOOL "Mbed TLS programs" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "Mbed TLS tests" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "Mbed TLS fatal warnings" FORCE)
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON CACHE BOOL "Mbed TLS install" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "Mbed TLS shared" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "Mbed TLS static" FORCE)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/include/claimer/mbedtls_user_config.h" CACHE FILEPATH "Pinned claimer crypto overrides" FORCE)
set(timestamp_args)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
  list(APPEND timestamp_args DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endif()
FetchContent_Declare(mbedtls
  URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.7/mbedtls-3.6.7.tar.bz2
  URL_HASH SHA256=a7e8bcbec0e6f761b4af24f25677626b35f762f68eef79c08677a363212d11f6
  ${timestamp_args}
  PATCH_COMMAND ${CMAKE_COMMAND} "-DMBEDTLS_SOURCE_DIR=<SOURCE_DIR>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/PatchMbedTLSCMakePolicy.cmake")
FetchContent_MakeAvailable(mbedtls)
set(MBEDTLS_SOURCE_DIR "${mbedtls_SOURCE_DIR}")
include(cmake/PatchMbedTLSP2CRSAPSS.cmake)
include(cmake/PatchMbedTLSP2CX509.cmake)
include(cmake/PatchMbedTLSReassembly.cmake)
include(cmake/PatchMbedTLSX25519.cmake)
unset(MBEDTLS_SOURCE_DIR)
foreach(component IN ITEMS mbedcrypto mbedx509 mbedtls)
  target_include_directories(${component} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_include_directories(${component} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/mbedtls-adapters")
  target_link_libraries(${component} PUBLIC Threads::Threads)
endforeach()
target_sources(mbedcrypto PRIVATE vendor/mbedtls-adapters/mbedtls_rsa_pss.c)
option(CLAIMER_X25519_ACCELERATION "Use bundled HACL for validated PSA X25519 multiplications" ON)
option(CLAIMER_X25519_TESTING "Instrument X25519 routing for regression tests only" OFF)
if(CLAIMER_X25519_ACCELERATION)
  target_compile_definitions(everest PUBLIC CONNECTCOIN_X25519_ACCELERATION)
  target_sources(mbedcrypto PRIVATE vendor/mbedtls-adapters/mbedtls_x25519.c)
  target_include_directories(mbedcrypto PRIVATE
    "${mbedtls_SOURCE_DIR}/3rdparty/everest/include/everest"
    "${mbedtls_SOURCE_DIR}/3rdparty/everest/include/everest/kremlib")
  if(CLAIMER_X25519_TESTING)
    target_compile_definitions(mbedcrypto PUBLIC CONNECTCOIN_X25519_TESTING)
  endif()
endif()
option(CLAIMER_SHA256_ACCELERATION "Dispatch SHA-224/256 compression to SHA-NI when supported" ON)
option(CLAIMER_SHA256_TESTING "Instrument SHA dispatch and allow thread-local fallback in tests only" OFF)
option(CLAIMER_SHA256_X86_BACKEND "Compile optional x86-64 SHA-NI backend (runtime CPU-gated)" ON)
if(CLAIMER_SHA256_ACCELERATION)
  target_compile_definitions(mbedcrypto PUBLIC CONNECTCOIN_SHA256_ACCELERATION)
  target_sources(mbedcrypto PRIVATE
    vendor/mbedtls-adapters/mbedtls_sha256.cpp
    vendor/mbedtls-adapters/mbedtls_sha256_portable.c)
  if(CLAIMER_SHA256_TESTING)
    target_compile_definitions(mbedcrypto PUBLIC CONNECTCOIN_SHA256_TESTING)
  endif()
  # CMAKE_SYSTEM_PROCESSOR can describe the host for Visual Studio cross-builds.
  # Check the actual C++ target, never execute a probe, and fail closed to the
  # portable implementation. Universal Apple binaries must not get x86 ISA flags.
  set(claimer_sha256_x86_target OFF)
  if(CLAIMER_SHA256_X86_BACKEND AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    if((NOT CMAKE_VS_PLATFORM_NAME OR CMAKE_VS_PLATFORM_NAME STREQUAL "x64") AND
       (NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES OR CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64"))
      include(CheckCXXSourceCompiles)
      check_cxx_source_compiles("
        #if defined(__aarch64__) || defined(_M_ARM64) || defined(_M_ARM64EC)
        #error SHA-NI is not an ARM target backend
        #endif
        #if !defined(__x86_64__) && !defined(_M_X64)
        #error SHA-NI requires an x86-64 compiler target
        #endif
        int main() { return 0; }
      " CLAIMER_SHA256_COMPILER_X86_64)
      set(claimer_sha256_x86_target "${CLAIMER_SHA256_COMPILER_X86_64}")
    endif()
  endif()
  if(claimer_sha256_x86_target)
    target_compile_definitions(mbedcrypto PRIVATE CONNECTCOIN_SHA256_X86_BACKEND)
    target_sources(mbedcrypto PRIVATE vendor/bitcoin-core/sha256_x86_shani.cpp)
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/vendor/bitcoin-core/sha256_x86_shani.cpp" TARGET_DIRECTORY mbedcrypto PROPERTIES
      COMPILE_DEFINITIONS "ENABLE_SSE41;ENABLE_X86_SHANI")
    if(NOT MSVC)
      # Never apply these ISA flags to the portable dispatcher or other code.
      set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/vendor/bitcoin-core/sha256_x86_shani.cpp" TARGET_DIRECTORY mbedcrypto PROPERTIES
        COMPILE_OPTIONS "-msha;-mssse3;-msse4.1")
    endif()
  endif()
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../helpers/p2c_roots_v1.pem" root_hash)
if(NOT root_hash STREQUAL "f66dff1bdf8f96060b8177976f8b7d9254bc89bc4db933d769f7384d28480bc9")
  message(FATAL_ERROR "Immutable P2C v1 root bundle hash mismatch")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../helpers/p2c_roots_v1.pem" CLAIMER_ROOT_HEX HEX)
# MSVC 2022 cannot compile the complete PEM as one string literal. Preserve
# every pinned input byte in an ordinary array, plus the PEM parser's NUL.
string(REGEX REPLACE "(................................)" "\\1\n" CLAIMER_ROOT_HEX "${CLAIMER_ROOT_HEX}")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," CLAIMER_ROOT_BYTES "${CLAIMER_ROOT_HEX}")
configure_file(roots.hpp.in generated/roots.hpp @ONLY)
add_library(claimer_crypto STATIC crypto.cpp)
target_include_directories(claimer_crypto PUBLIC include PRIVATE vendor/mbedtls-adapters "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(claimer_crypto PUBLIC mbedtls mbedx509 mbedcrypto Threads::Threads)
if(WIN32)
  target_link_libraries(claimer_crypto PUBLIC bcrypt)
endif()
add_library(claimer_json STATIC vendor/univalue/lib/univalue.cpp vendor/univalue/lib/univalue_get.cpp vendor/univalue/lib/univalue_read.cpp vendor/univalue/lib/univalue_write.cpp)
target_include_directories(claimer_json PUBLIC vendor/univalue/include)
option(CLAIMER_CRYPTO_ONLY "Build only crypto regression tests" OFF)
if(NOT CLAIMER_CRYPTO_ONLY)
  add_library(claimer_network STATIC network.cpp p2c_tls_private.c)
  target_include_directories(claimer_network PUBLIC include PRIVATE "${mbedtls_SOURCE_DIR}/library")
  target_link_libraries(claimer_network PUBLIC claimer_crypto)
  if(WIN32)
    target_link_libraries(claimer_network PUBLIC ws2_32)
  endif()
  add_library(claimer_service STATIC service.cpp)
  target_link_libraries(claimer_service PUBLIC claimer_network claimer_json)
  add_executable(connectcoin-claims main.cpp)
  set_target_properties(connectcoin-claims PROPERTIES OUTPUT_NAME connectwallet-claims)
  target_link_libraries(connectcoin-claims PRIVATE claimer_service)
  install(TARGETS connectcoin-claims RUNTIME DESTINATION . COMPONENT claimerRuntime)
  install(FILES LICENSE PROVENANCE.md THIRD-PARTY-NOTICES.txt DESTINATION licenses COMPONENT claimerRuntime)
  install(FILES vendor/CORE-LICENSE DESTINATION licenses COMPONENT claimerRuntime)
  install(FILES vendor/bitcoin-core/COPYING DESTINATION licenses RENAME Bitcoin-Core-COPYING COMPONENT claimerRuntime)
  install(FILES "${mbedtls_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME MbedTLS-LICENSE COMPONENT claimerRuntime)
  install(FILES vendor/univalue/COPYING DESTINATION licenses RENAME UniValue-COPYING COMPONENT claimerRuntime)
endif()

    execute_process(COMMAND "${CLAIMER_TEST_PYTHON}" -c "import cryptography" RESULT_VARIABLE fixture_dependencies OUTPUT_QUIET ERROR_QUIET)
    if(fixture_dependencies EQUAL 0)
      add_test(NAME native_network_fixture COMMAND "${CLAIMER_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/test_network_fixture.py" --binary $<TARGET_FILE:claimer-network-tests>)

      add_test(NAME native_stdio_pipe COMMAND "${CLAIMER_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/test_stdio_pipe.py" --binary $<TARGET_FILE:connectcoin-claims>)
      set_tests_properties(native_stdio_pipe PROPERTIES TIMEOUT 30)
    else()
      message(WARNING "Offline native TLS fixtures require CLAIMER_TEST_PYTHON with cryptography; no fixture test registered")
    endif()
  endif()
endif()
